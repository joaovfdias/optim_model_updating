!CODIGO EXEMPLO_GIEDRE
!PENDENTE:
! 1 - ADICIONAR OS 4 APOIOS FLEXÍVEIS   CHECK
! 2 - COLOCAR A PARTE DO 'SOLVE'   check
! 3 - AJUSTAR NOMES DE PARÂMETROS A OTIMIZAR (MÓDULO E DENSIDADE DO CONCRETO E RIGIDEZ DOS APOIOS)(a fazer)
! 4 - PREPARAR SAÍDA DE DADOS    (check)

/PREP7

! BRIDGE DIMENSIONS
L_PONTE = 25
B_PONTE = 8
T_LAJE = .3
DIST_1 = 1.5      ! DISTANCIA ENTRE PONTOS DED VERIFICACAO DE DESLOCAMENTO
H_VIGA = 2
B_VIGA = .6
modElast = %modulo%   ! Módulo de elasticidade (Pa) - variável a ser substituída
poisson = %poisson%   ! Coeficiente de Poisson
dens = %dens%         ! Densidade do material (kg/m³)
k1 = %rigidez1%       ! Rigidez 1
k2 = %rigidez2%       ! Rigidez 2
k3 = %rigidez3%      ! Rigidez 3
k4 = %rigidez4%    ! Rigidez 4
NUMFREQ = 20 ! número de frequências
NUMPONTOS = 12   ! malha para MAC

! ELEMENT TYPE DEFINITION
ET,1,188  ! Beam
ET,2,181  ! Shell
ET,3,14   ! Spring
keyopt,3,2,2  ! Defines the spring degree of freedom to 1D in UY
keyopt,3,3,0  ! Stes the spring as a longitudinal spring-damper for 3D elements

! MATERIAL PROPRIETIES DEFINITION
MP,EX,, modElast
MP,PRXY,,poisson
MP,DENS,,dens

! GEOMETRY DRAWING

K,1,0,0,-B_PONTE/2
K,2,L_PONTE,0,-B_PONTE/2
K,3,L_PONTE,0,B_PONTE/2
K,4,0,0,B_PONTE/2

A,1,2,3,4

! SUPPORT BEAM LINES
WPOFFS,,,-B_PONTE/2+DIST_1
ASBW,ALL    ! DIVIDES THE AREA
WPCSYS,0
WPOFFS,,,B_PONTE/2-DIST_1    !B_PONTE-2*DIST_1         (ESTAVA ASSIM ANTES E EU AJUSTEI)
ASBW,ALL
WPCSYS,0

! Verifica e mescla linhas duplicadas
NUMCMP,LINE
NUMCMP,KP
NUMMRG,KP



! KEYPOINTS TO THE BEAM ORIENTATION
K,20,,1, -B_PONTE/2+DIST_1
K,21,,1,B_PONTE/2-DIST_1

! DEFINITION OF BEAM CROSS-SECTION
SECTYPE,1,BEAM,RECT,'VIGA',3
SECDATA,B_VIGA,H_VIGA,3,3
SECOFFSET,USER,0,H_VIGA/2    ! Beam node will be offset to the location specified by HALF OF THE L OF THE BEAM

! DEFINITION OF SLAB SECTION
SECTYPE,2,SHELL,,'LAJE',3
SECDATA,T_LAJE
SECOFFSET,TOP
!! seccontrol,,,,added_mass  QUER-SE ADICIONAR MASSA EM TODOS OS ELEMENTOS?

! MESH ATTRIBUTES DEFINITION
lesize,all,.5
aatt,1,,2,,2     !Attributes the material 1, ET 2(shell) and section 2(shell and t) to all areas
amesh,all
lsel,s,loc,z,-B_PONTE/2+DIST_1
latt,1,,1,,20,,1      ! Attributes material 1, ET1, keypoint 21 and  SECTION1(rect beam)to selected line
lmesh,all
lsel,all
lsel,s,loc,z,B_PONTE/2-DIST_1
latt,1,,1,,21,,1 ! Attributes material 1, ET1, keypoint 21 and  SECTION1(rect beam)to selected line
lmesh,all
ALLSEL

NUMCMP,NODE
NUMMRG,NODE     ! Tolerância bem pequena
NUMMRG,ELEM
NUMMRG,ELEM

! APPLY DISPLACEMENT RESTRICTIONS ON THE SLAB SUPPORTS (x, y, z translation)
ksel,s,loc,x,0
ksel,a,loc,x,l_ponte
cm,aux1,kp
ksel,s,loc,z,-B_PONTE/2+DIST_1
ksel,a,loc,z,B_PONTE/2-DIST_1
ksel,r,loc,y,0
cmsel,r,aux1
nslk,s
d,all,ux
d,all,uz
allsel



! REAL CONSTANTS FOR THE STIFFNESS VALUES OF THE MATERIALS
r,1,
r,2,k1
r,3,k2
r,4,k3
r,5,k4

! STIFFNESS MATERIALS CREATION
type,3

real,2  ! Related to K1
N,,0,-1,-B_PONTE/2 + DIST_1
e,node(0,-1,-B_PONTE/2 + DIST_1),node(0,0,-B_PONTE/2 + DIST_1)

real,3   ! Related to K2
N,,L_PONTE,-1,-B_PONTE/2 + DIST_1
e,node(L_PONTE,-1,-B_PONTE/2 + DIST_1),node(L_PONTE,0,-B_PONTE/2 + DIST_1)

real,4   ! Related to K3
N,,L_PONTE,-1,B_PONTE/2 - DIST_1
e,node(L_PONTE,-1,B_PONTE/2 - DIST_1),node(L_PONTE,0,B_PONTE/2 - DIST_1)

real,5   ! Related to K4
N,,0,-1,B_PONTE/2 - DIST_1
e,node(0,-1,B_PONTE/2 - DIST_1),node(0,0,B_PONTE/2 - DIST_1)

! STIFFNESS ELEMENTS DISPLACEMENT
nsel,s,loc,y,-1
d,all,all

ALLSEL

! SOLUÇÃO


/solu
antype,modal
modopt,lanb,NUMFREQ
solve

! EXPORTAÇÃO DOS RESULTADOS
/post1


! DEFINIÇÃO DOS PONTOS
*dim, Nos, array, NUMPONTOS
aux = 0

*do, g, 1, 4  ! Posição em z    VERIFICAR DAQUI EM DIANTE
    z_pos = -B_PONTE/2 +(g-1)*(B_PONTE/3)
    *do, J, 1, 3    !  Posição em x
        aux = aux + 1
        x_pos = L_PONTE/4*(j)
        Nos(aux) = node(x_pos, 0, z_pos)
    *enddo
*enddo

! FREQUÊNCIAS E DESLOCAMENTOS
*dim, Freq, array, NUMFREQ
*dim, Modos, array, NUMPONTOS, NUMFREQ  ! Matriz para deslocamentos Y

*do, i, 1, NUMFREQ
    set,,i
    *get, Freq(i), ACTIVE, 0, SET, FREQ  ! Armazena frequência

    *do, j, 1, NUMPONTOS
        Modos(j,i) = uy(Nos(j))        ! Deslocamento em Y
    *enddo
*enddo

! GERAR ARQUIVOS

*CREATE, scratch, gui

*cfopen, out_freq, txt
*VWRITE, Freq(1)
(f12.5)
*cfclose

*MWRITE,Modos,out_modes,txt
(f8.5)


*END
/INPUT, scratch, gui
